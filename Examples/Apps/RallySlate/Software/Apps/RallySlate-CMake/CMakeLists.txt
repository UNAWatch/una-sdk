cmake_minimum_required(VERSION 3.21)
set(APP_NAME "RallySlate")
set(APP_TYPE "Utility")
set(DEV_ID "brodyga45")
# Development-only default. For App Store releases use the portal-generated ID.
set(APP_ID "BEC65A5C096E4894" CACHE STRING "UNA portal-generated 16-digit hexadecimal app ID")
string(LENGTH "${APP_ID}" APP_ID_LENGTH)
if(NOT APP_ID_LENGTH EQUAL 16 OR NOT APP_ID MATCHES "^[0-9A-Fa-f]+$")
  message(FATAL_ERROR "APP_ID must contain exactly 16 hexadecimal digits")
endif()
set(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(LIBS_PATH "${APP_ROOT}/Service")
set(GUI_PATH "${APP_ROOT}/Source")
set(RESOURCES_PATH "${APP_ROOT}/Resources")
set(OUTPUT_PATH "${CMAKE_CURRENT_BINARY_DIR}")
file(MAKE_DIRECTORY ${OUTPUT_PATH})
include($ENV{UNA_SDK}/cmake/una-app.cmake)
include($ENV{UNA_SDK}/cmake/una-sdk.cmake)
project(RallySlate C CXX)
set(BUILD_VERSION "0.1.0")
set(SERVICE_SOURCES ${APP_ROOT}/Service/Service.cpp ${UNA_SDK_SOURCES_COMMON} ${UNA_SDK_SOURCES_SERVICE})
set(SERVICE_INCLUDE_DIRS ${APP_ROOT}/Service ${UNA_SDK_INCLUDE_DIRS_COMMON} ${UNA_SDK_INCLUDE_DIRS_SERVICE})
una_app_build_service(RallySlateService.elf)
set(GUI_SOURCES ${APP_ROOT}/Source/Gui.cpp ${APP_ROOT}/Source/Model.cpp ${APP_ROOT}/Source/Screen.cpp ${APP_ROOT}/Source/Notices.cpp
  ${UNA_SDK_SOURCES_COMMON} ${UNA_SDK_SOURCES_GUI_LVGL})
set(GUI_INCLUDE_DIRS ${APP_ROOT} ${APP_ROOT}/Source ${UNA_SDK_INCLUDE_DIRS_COMMON}
  ${UNA_SDK_INCLUDE_DIRS_GUI_LVGL} ${UNA_SDK_INCLUDE_DIRS_SERVICE})
set(GUI_COMPILE_DEFINITIONS ${UNA_SDK_DEFINES_GUI_LVGL} LV_MEM_SIZE=65536)
set(GUI_STACK_SIZE "24*1024")
una_app_build_gui(RallySlateGUI.elf)
una_app_build_app()
